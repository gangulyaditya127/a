version: "3.9"

services:
  redis-stack:
    image: redis/redis-stack-server:latest
    container_name: redis-stack
    mem_limit: 512m
    ports:
      - "6379:6379"
    command: redis-stack-server --maxmemory 512mb --maxmemory-policy noeviction
    restart: unless-stopped
    volumes:
      - redis_data:/data

  scylladb:
    image: scylladb/scylla:latest
    container_name: scylladb
    mem_limit: 3g
    cpuset: "0-2"
    ports:
      - "9042:9042"
    command: --smp 3 --memory 2G --overprovisioned 1 --developer-mode 1
    restart: unless-stopped
    volumes:
      - scylla_data:/var/lib/scylla

volumes:
  redis_data:
  scylla_data:
