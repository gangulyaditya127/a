@app.post("/upload/nexp-site-details/")
def upload_va_grouping_title(file: UploadFile = File(...)):
    target_path = "Nexpose site deatils and spocs.xlsx"
    try:
        with open(target_path, "wb") as buffer:
            shutil.copyfileobj(file.file, buffer)
        return {
            "message": "Nexpose site deatils and spocs uploaded successfully",
            "saved_as": target_path
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
