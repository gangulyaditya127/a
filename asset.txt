@app.get("/download-combined-excel")
def download_combined_excel():

    if not os.path.exists(EXCEL_DATA_DIR):
        raise HTTPException(404, "excel_data directory not found")

    excel_files = [
        f for f in os.listdir(EXCEL_DATA_DIR)
        if f.endswith(".xlsx")
    ]

    if not excel_files:
        raise HTTPException(404, "No Excel files found in directory")

    # ✅ Output file
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    output_file = f"Combined_Excel_{timestamp}.xlsx"
    output_path = os.path.join(EXCEL_DATA_DIR, output_file)

    try:
        with pd.ExcelWriter(output_path, engine="openpyxl") as writer:

            for file in excel_files:
                file_path = os.path.join(EXCEL_DATA_DIR, file)

                try:
                    df = pd.read_excel(file_path)

                    # ✅ Sheet name = file name (max 31 chars Excel limit)
                    sheet_name = os.path.splitext(file)[0][:31]

                    df.to_excel(writer, sheet_name=sheet_name, index=False)

                except Exception as e:
                    print(f"Skipping file {file}: {str(e)}")

        return FileResponse(
            path=output_path,
            filename=output_file,
            media_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
        )

    except Exception as e:
        raise HTTPException(500, f"Failed to combine files: {str(e)}")
