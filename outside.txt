@app.post("/generate-nexpose-ip-excel")
def generate_nexpose_ip_excel(db=Depends(get_db)):
    logger.info("=== Generate Nexpose Excel API called ===")

    try:
        # Validate reference files
        if not os.path.exists(REFERENCE_FILE):
            raise FileNotFoundError("Reference Excel file not found")
        
        if not os.path.exists(SITE_ID_FILE):
            raise FileNotFoundError("Site ID Excel file not found")

        site_df = pd.read_excel(SITE_ID_FILE)

        if SITE_ID_COLUMN not in site_df.columns:
            raise ValueError(f"Column '{SITE_ID_COLUMN}' not found in site ID Excel file")

        SITE_IDS = (
            site_df[SITE_ID_COLUMN]
            .dropna()
            .astype(int)
            .unique()
            .tolist()
        )

        if not SITE_IDS:
            raise ValueError("No Site IDs found in site ID Excel file")

        # Reference Excel for mapping
        ref_df = pd.read_excel(REFERENCE_FILE)

        all_rows = []
        successful_sites = []
        failed_sites = []
        empty_data_sites = []

        # Static Site IDs for now
        SITE_IDS = [191, 192, 193, 194]

        for site_id in SITE_IDS:
            logger.info(f"Processing Site ID: {site_id}")

            try:
                site_row = ref_df[ref_df["Site- ID"] == site_id]
                site_name = site_row.iloc[0]["Site-Name"] if not site_row.empty else f"Site-{site_id}"
                owner = site_row.iloc[0]["Owner"] if not site_row.empty else "Unknown"

                creds = get_active_nexpose_credentials(db)
                base_url = creds["base_url"]
                api_base = base_url.rstrip("/") + "/api/3/sites"

                response = requests.get(
                    f"{api_base}/{site_id}/included_targets",
                    auth=(creds["username"], creds["password"]),
                    verify=False,
                    timeout=60
                )

                response.raise_for_status()
                raw_addresses = response.json().get("addresses", [])

                if not raw_addresses:
                    empty_data_sites.append(site_id)
                    continue

                expanded_ips = expand_ip_ranges(raw_addresses)

                for ip in expanded_ips:
                    all_rows.append({
                        "Defind IP": ip,
                        "Site ID": site_id,
                        "Site Name": site_name,
                        "Owner": owner
                    })

                successful_sites.append(site_id)

            except Exception as e:
                failed_sites.append({"site_id": site_id, "error": str(e)})
                logger.error(f"Error processing site {site_id}: {e}")

        if not all_rows:
            raise ValueError("No IPs collected from provided Site IDs")

        df = pd.DataFrame(all_rows)
        df.drop_duplicates(subset=["Defind IP"], inplace=True)

        # ================================================
        # ONLY UPDATE Nexpose- raw data.xlsx (no new file)
        # ================================================
        raw_data_path = os.path.join("reference_excel", "Nexpose- raw data.xlsx")

        if os.path.exists(raw_data_path):
            with pd.ExcelWriter(raw_data_path, engine="openpyxl", mode="a", if_sheet_exists="replace") as writer:
                df.to_excel(writer, sheet_name="Def -IP", index=False)
        else:
            with pd.ExcelWriter(raw_data_path, engine="openpyxl") as writer:
                df.to_excel(writer, sheet_name="Def -IP", index=False)

        logger.info("Sheet 'Def -IP' updated in Nexpose- raw data.xlsx")

        return {
            "status": "partial_success" if failed_sites or empty_data_sites else "success",
            "updated_file": raw_data_path,
            "ip_count": len(df),
            "site_summary": {
                "total": len(SITE_IDS),
                "successful": successful_sites,
                "failed": failed_sites,
                "no_data": empty_data_sites
            },
            "data": df.to_dict(orient="records")
        }

    except Exception as e:
        logger.exception("Failed to generate Nexpose Excel")
        raise HTTPException(status_code=500, detail=str(e))

                verify=False
            )

                continue

# 🚨 CHECK FOR RUNNING OR INTEGRATING SCANS — STOP IMMEDIATELY

            running_scans = [
                s for s in scans_list
                if s.get("status") in ["running", "integrating"]
            ]

            if running_scans:
                latest_running = max(running_scans, key=lambda x: x.get("startTime"))
                return {
                    "status": "error",
                    "site_id": site_id,
                    "scan_id": latest_running.get("id"),

            # 🔹 Filter Scheduled + Finished scans
            # -----------------------------------------------------------
            valid_scans = [
                s for s in scans_list

            ]

            if not valid_scans:
                continue   # Option A: skip this site

            # -----------------------------------------------------------
            # 🔹 Pick latest scan
            # -----------------------------------------------------------
            latest_scan = max(valid_scans, key=lambda x: x.get("endTime"))

            start_time = latest_scan.get("startTime")
            end_time = latest_scan.get("endTime")

                "start_time": start_time,

            # 4️⃣ Create Scan Report Payload
