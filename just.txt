# --- Pydantic Models ---
class ChatMessage(BaseModel):
    role: str = Field(..., description="Role of the sender: 'user' or 'model'")
    text: str = Field(..., description="The message content")


class ChatRequest(BaseModel):
    provider: str = Field(default="AWS", description="Cloud provider context e.g. AWS, Azure, GCP")
    message: str = Field(..., description="The latest user message")
    history: List[ChatMessage] = Field(default_factory=list, description="Previous conversation history")


class ProposedResource(BaseModel):
    service: str
    count: int
    configDetails: str


class ChatResponse(BaseModel):
    reply: str = Field(..., description="The conversational reply from the AI cleaned of JSON")
    proposed_resources: Optional[List[ProposedResource]] = Field(
        default_factory=list,
        description="Parsed resource recommendations"
    )


# --- Helper Functions ---
def get_system_prompt(provider: str) -> str:
    return f"""
You are an expert Cloud Solution Architect specializing in {provider}.
Your goal is to help the user design their cloud infrastructure and provision the right resources.

Follow these rules strictly:
1. Have a natural, conversational interaction.
2. Ask 1-2 clarifying questions at a time to understand their workload.
3. Do not overwhelm the user with too many questions at once.
4. Once you have gathered enough requirements, propose a final cloud resource configuration.
5. IMPORTANT: When proposing the configuration, you MUST include a JSON array containing the resources.
6. Wrap the JSON exactly in a ```json block.

Schema for the JSON array:
[
  {{
    "service": "Name of the cloud service",
    "count": number,
    "configDetails": "Specific configuration details e.g. t3.medium, Multi-AZ"
  }}
]

Always provide a friendly natural language explanation alongside the JSON block.
"""


def build_llm_prompt(request: ChatRequest) -> str:
    """
    Builds a single prompt string for llm.invoke().
    Since the new LLM call uses invoke(llm_prompt), we combine:
    - system instruction
    - conversation history
    - latest user message
    """

    system_prompt = get_system_prompt(request.provider)

    history_text = ""
    if request.history:
        history_text = "\nConversation history:\n"
        for msg in request.history:
            role = "User" if msg.role.lower() == "user" else "Assistant"
            history_text += f"{role}: {msg.text}\n"

    prompt = f"""
{system_prompt}

{history_text}

Latest user message:
User: {request.message}

Assistant:
"""

    return prompt.strip()


def extract_response_text(llm_response: Any) -> str:
    """
    Handles different possible response formats from LangChain-style LLMs.
    llm.invoke() may return:
    - string
    - object with .content
    - dict-like response
    """

    if isinstance(llm_response, str):
        return llm_response

    if hasattr(llm_response, "content"):
        return llm_response.content

    if isinstance(llm_response, dict):
        if "content" in llm_response:
            return llm_response["content"]
        if "text" in llm_response:
            return llm_response["text"]
        if "response" in llm_response:
            return llm_response["response"]

    return str(llm_response)


def extract_json_resources(text: str) -> tuple[str, List[Dict[str, Any]]]:
    """
    Extracts JSON resources from the LLM output and returns:
    - cleaned natural language reply
    - parsed resource list
    """

    json_regex = r"```json\s*(.*?)\s*```"
    matches = re.findall(json_regex, text, re.DOTALL)

    resources = []
    clean_text = text

    if matches:
        try:
            resources = json.loads(matches[-1])
            clean_text = re.sub(json_regex, "", text, flags=re.DOTALL).strip()
        except json.JSONDecodeError:
            logger.warning("Failed to parse JSON block from LLM response.")

    return clean_text, resources


# --- Endpoints ---
@app.post("/chat", response_model=ChatResponse, tags=["Chat"])
async def generate_chat_response(request: ChatRequest):
    try:
        global llm

        if llm is None:
            llm = initialize_llm()

        llm_prompt = build_llm_prompt(request)

        llm_response = llm.invoke(llm_prompt)

        response_text = extract_response_text(llm_response)

        clean_text, resources = extract_json_resources(response_text)

        return ChatResponse(
            reply=clean_text,
            proposed_resources=resources
        )

    except Exception as e:
        logger.error(f"Chat generation failed: {str(e)}")
        logger.error(traceback.format_exc())
        raise HTTPException(status_code=500, detail=str(e))
