@app.post("/owner-mapping")
def add_owner_mapping(
    payload: OwnerMappingCreate,
    db=Depends(get_db)
):
    exists = db.query(OwnerMapping).filter(
        OwnerMapping.owner_key == payload.owner_key
    ).first()

    if exists:
        raise HTTPException(status_code=400, detail="Owner key already exists")

    record = OwnerMapping(
        owner_key=payload.owner_key,
        to_emails=payload.to_emails,
        cc_emails=payload.cc_emails
    )

    db.add(record)
    db.commit()

    return {"message": "Owner mapping added successfully"}


@app.delete("/owner-mapping/{owner_key}")
def delete_owner_mapping(owner_key: str, db=Depends(get_db)):
    record = db.query(OwnerMapping).filter(
        OwnerMapping.owner_key == owner_key
    ).first()

    if not record:
        raise HTTPException(status_code=404, detail="Owner key not found")

    db.delete(record)
    db.commit()

    return {"message": "Owner mapping deleted"}


@app.get("/owner-mapping", response_model=list[OwnerMappingResponse])
def list_owner_mappings(db=Depends(get_db)):
    records = db.query(OwnerMapping).all()

    return [
        {
            "owner_key": r.owner_key,
            "to_emails": r.to_emails,
            "cc_emails": r.cc_emails
        }
        for r in records
    ]
