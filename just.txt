# =========================
# Load Excel
# =========================
file_path = "SOE sample (1).xlsx"

df = pd.read_excel(file_path, sheet_name="One engine", header=2)

# Clean column names
df.columns = df.columns.astype(str).str.strip()

# Auto-detect columns
def find_column(df, keyword):
    for col in df.columns:
        if keyword in col.lower():
            return col
    raise Exception(f"Column containing '{keyword}' not found")

source_col = find_column(df, "source")
dest_col = find_column(df, "destination")
port_col = find_column(df, "port")
protocol_col = find_column(df, "protocol")

print("✅ Columns:", source_col, dest_col, port_col, protocol_col)

# Handle merged cells
df[source_col] = df[source_col].replace("", pd.NA).ffill()
df[dest_col] = df[dest_col].replace("", pd.NA).ffill()

# =========================
# Load Subnet Sheet
# =========================
subnet_df = pd.read_excel(
    file_path,
    sheet_name="Ultimatix Subnets",
    header=[0, 1, 2]
)

# Flatten columns
subnet_df.columns = [
    f"{str(a).strip()}|{str(b).strip()}|{str(c).strip()}"
    for a, b, c in subnet_df.columns
]

# =========================
# Helper Functions
# =========================

def normalize(text):
    return str(text).strip().lower().replace("-", " ").replace("_", " ")

def parse_ports(val):
    if pd.isna(val):
        return []

    result = []
    for v in str(val).split(","):
        v = v.strip()

        if "-" in v:
            try:
                start, end = v.split("-")
                result.extend([str(p) for p in range(int(start), int(end) + 1)])
            except:
                continue
        elif v:
            result.append(v)

    return result


def is_ip_or_subnet(value):
    return bool(re.match(r"\d+\.\d+\.\d+\.\d+", str(value)))


def expand_subnet(subnet):
    try:
        network = ipaddress.ip_network(subnet, strict=False)

        if network.num_addresses > 1024:
            return [subnet]

        return [str(ip) for ip in network]
    except:
        return [subnet]


def resolve_value(value):
    """
    ✅ Handles:
    - IPs / CIDR
    - Ultimatix groups
    - multi-env
    - PR / DR filter
    - complex input formats
    """
    if pd.isna(value):
        return []

    value_str = str(value).strip()
    final_list = []

    parts = [v.strip() for v in value_str.split(",") if v.strip()]

    # ✅ Case 1: pure IP/subnet
    if all(is_ip_or_subnet(v) for v in parts):
        for v in parts:
            final_list.extend(expand_subnet(v))
        return final_list

    v_clean = normalize(value_str)

    # ✅ Detect tier
    tier = None
    if "non prod" in v_clean:
        tier = "non prod"
    elif "prod" in v_clean:
        tier = "prod"

    # ✅ Detect PR / DR
    region_filter = []
    if "pr" in v_clean:
        region_filter.append("pr")
    if "dr" in v_clean:
        region_filter.append("dr")

    # ✅ Detect environments
    envs = []

    if "on prem" in v_clean:
        envs.append("on prem")
    if "azure" in v_clean:
        envs.append("azure")
    if "aws" in v_clean:
        envs.append("aws")

    # ✅ fallback parsing for comma-only formats
    if not envs and "," in value_str:
        for p in value_str.split(","):
            p = p.lower()
            if "on prem" in p:
                envs.append("on prem")
            elif "azure" in p:
                envs.append("azure")
            elif "aws" in p:
                envs.append("aws")

    if not envs or not tier:
        return []

    # ✅ fetch subnet values
    for env in envs:
        for col in subnet_df.columns:
            parts = col.lower().split("|")

            if len(parts) != 3:
                continue

            col_env, col_tier, col_region = parts

            if env in col_env and tier in col_tier:

                # ✅ apply PR/DR filtering
                if region_filter:
                    if not any(r in col_region for r in region_filter):
                        continue

                subnets = subnet_df[col].dropna().astype(str).tolist()

                for s in subnets:
                    final_list.extend(expand_subnet(s))

    return final_list


# =========================
# Insert Data
# =========================
insert_query = """
INSERT INTO network_rules (source, destination, port_no, protocol)
VALUES (%s, %s, %s, %s)
"""

seen = set()

for _, row in df.iterrows():

    sources = resolve_value(row[source_col])
    destinations = resolve_value(row[dest_col])

    ports = parse_ports(row[port_col])
    protocol = str(row[protocol_col]).strip()

    for src in sources:
        for dst in destinations:
            for p in ports:

                record = (src, dst, p, protocol)

                if record not in seen:
                    seen.add(record)
                    cur.execute(insert_query, record)

conn.commit()

cur.close()
conn.close()
